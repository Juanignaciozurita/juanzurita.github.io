Let’s define function

def ces_func(k,l,A=1.5,alpha=0.33,sigma=1.5):
    if sigma==1:
        print('function takes Cobb-Douglas form')
        y = A*k**(alpha)*l**(1-alpha)

    else:
        y = A*(alpha*k**((sigma-1)/sigma) +(1-alpha)*l**((sigma-1)/sigma) )**(sigma/(sigma-1))
        
    return y
