\begin{frame}{Why Convexity?}
\begin{block}{The Convex Optimisation Guarantee}Every local minimum is \alert{global}. Efficient algorithms exist.\end{block}\par\bigskip Many economic models are naturally convex: cost minimisation, portfolio optimisation, regression.
\end{frame}

\begin{frame}{Lagrangian Duality}
\textbf{Primal:} $\min f(\mathbf{x})$ s.t.\ $g_i \leq 0$\\\bigskip\textbf{Dual function:} $d(\boldsymbol{\mu}) = \inf_\mathbf{x}\{f(\mathbf{x}) + \sum\mu_i g_i(\mathbf{x})\}$\\\bigskip\textbf{Weak duality:} $d^* \leq p^*$ always\\\textbf{Strong duality:} $d^* = p^*$ for convex problems (Slater's condition)
\end{frame}

\begin{frame}[fragile]{CVXPY}
CVXPY: a Python library for \textbf{disciplined convex programming}.

\begin{lstlisting}
import cvxpy as cp
x = cp.Variable(n)
prob = cp.Problem(cp.Minimize(cp.sum_squares(A@x-b)),
                  [x >= 0])
prob.solve()
\end{lstlisting}

Automatically checks convexity and selects the right solver.
\end{frame}

\begin{frame}{Summary}
\begin{itemize}\item Convexity guarantees global optimality\item Strong duality: primal value = dual value\item CVXPY makes convex optimisation accessible in Python\item \textbf{Next week:} linear programming --- a special case with rich structure\end{itemize}
\end{frame}
