\begin{frame}{The Constrained Problem}
\textbf{Maximise} $f(\mathbf{x})$ \textbf{subject to} $g(\mathbf{x}) = 0$\\\bigskip\textbf{Lagrangian:} $\mathcal{L}(\mathbf{x},\lambda) = f(\mathbf{x}) - \lambda\,g(\mathbf{x})$\\\bigskip\textbf{FOC:} $\nabla f = \lambda\nabla g$ and $g(\mathbf{x}) = 0$\\\bigskip Geometrically: at the optimum, the level curve of $f$ is \alert{tangent} to the constraint.
\end{frame}

\begin{frame}{Utility Maximisation}
$\max\; u(x_1,x_2) = x_1^\alpha x_2^{1-\alpha}$ s.t.\ $p_1x_1 + p_2x_2 = m$\\\bigskip Marshallian demands: $x_1^* = \frac{\alpha m}{p_1}$, $x_2^* = \frac{(1-\alpha)m}{p_2}$\\\bigskip The multiplier $\lambda^*$ = marginal utility of income.
\end{frame}

\begin{frame}{The Envelope Theorem}
\begin{block}{Envelope Theorem}$\frac{\partial V}{\partial m} = \lambda^*$\end{block}\par\bigskip The Lagrange multiplier measures the \alert{shadow value} of the constraint.\\\bigskip An extra \$1 of income raises utility by exactly $\lambda^*$.
\end{frame}

\begin{frame}{Bordered Hessian}
To verify a constrained maximum, check the \textbf{bordered Hessian}:\\ $$\bar{H} = \begin{pmatrix} 0 & g_{x_1} & g_{x_2} \\\\ g_{x_1} & \mathcal{L}_{x_1x_1} & \mathcal{L}_{x_1x_2} \\\\ g_{x_2} & \mathcal{L}_{x_2x_1} & \mathcal{L}_{x_2x_2}\end{pmatrix}$$\\\bigskip For a maximum with one constraint: $|\bar{H}| > 0$.
\end{frame}

\begin{frame}{Summary}
\begin{itemize}\item Lagrange multipliers handle equality constraints\item $\lambda^*$ = shadow price of the constraint\item The envelope theorem connects multipliers to comparative statics\item \textbf{Next week:} inequality constraints and KKT conditions\end{itemize}
\end{frame}
