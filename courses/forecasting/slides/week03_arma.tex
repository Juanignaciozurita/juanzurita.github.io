\section{The AR Process}

\begin{frame}{AR(1) --- The Simplest Dynamic Model}
\[
y_t = c + \phi y_{t-1} + \varepsilon_t, \quad \varepsilon_t \sim WN(0, \sigma^2)
\]

\begin{itemize}
    \item \textbf{Stationary} iff $|\phi| < 1$
    \item Unconditional mean: $\mu = c / (1 - \phi)$
    \item Unconditional variance: $\sigma_y^2 = \sigma^2 / (1 - \phi^2)$
    \item ACF: $\rho_k = \phi^k$ (geometric decay)
    \item PACF: cuts off after lag 1
\end{itemize}
\end{frame}

\begin{frame}{AR(p) --- General Autoregressive}
\[
y_t = c + \phi_1 y_{t-1} + \phi_2 y_{t-2} + \cdots + \phi_p y_{t-p} + \varepsilon_t
\]

\textbf{Stationarity condition:} all roots of $1 - \phi_1 z - \phi_2 z^2 - \cdots - \phi_p z^p = 0$ lie \alert{outside} the unit circle.

\vspace{0.5em}
\textbf{PACF} of AR($p$): cuts off after lag $p$.
\end{frame}

\section{The MA Process}

\begin{frame}{MA(q) --- Moving Average}
\[
y_t = \mu + \varepsilon_t + \theta_1 \varepsilon_{t-1} + \cdots + \theta_q \varepsilon_{t-q}
\]

\begin{itemize}
    \item \alert{Always stationary} (finite linear combination of white noise)
    \item ACF: cuts off after lag $q$
    \item PACF: decays geometrically
    \item Invertibility requires roots of $1 + \theta_1 z + \cdots$ outside unit circle
\end{itemize}
\end{frame}

\section{ARMA Models}

\begin{frame}{ARMA(p,q)}
\[
y_t = c + \sum_{i=1}^{p}\phi_i y_{t-i} + \varepsilon_t + \sum_{j=1}^{q}\theta_j \varepsilon_{t-j}
\]

\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{Model} & \textbf{ACF} & \textbf{PACF} \\
\midrule
AR($p$) & Decays & Cuts off at $p$ \\
MA($q$) & Cuts off at $q$ & Decays \\
ARMA($p,q$) & Decays & Decays \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\begin{frame}{The Box-Jenkins Methodology}
\begin{enumerate}
    \item \textbf{Identification}: use ACF/PACF to determine $p, q$
    \item \textbf{Estimation}: fit by MLE (or conditional least squares)
    \item \textbf{Diagnostic checking}: are residuals white noise?
    \item \textbf{Forecasting}: generate $h$-step-ahead predictions
\end{enumerate}

\vspace{0.5em}
Modern practice also uses \textbf{AIC/BIC} for step 1.
\end{frame}

\begin{frame}{Summary}
\begin{itemize}
    \item \textbf{AR}: past \emph{values} determine the present
    \item \textbf{MA}: past \emph{shocks} determine the present
    \item \textbf{ARMA}: combines both
    \item ACF/PACF patterns $\Rightarrow$ model identification
    \item Next week: \alert{estimation, forecasting, and diagnostics}
\end{itemize}
\end{frame}
