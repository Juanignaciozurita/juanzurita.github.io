\section{Motivation}

\begin{frame}{The Curse of Dimensionality}
Central banks track \alert{hundreds} of economic indicators. \\
How to use all this information for forecasting?

\vspace{0.5em}
\textbf{Solution:} a small number of \alert{latent factors} drive the co-movement.
\[
y_{it} = \lambda_i' \mathbf{f}_t + e_{it}, \quad i = 1, \ldots, N
\]
\begin{itemize}
    \item $\mathbf{f}_t$: $k$-dimensional factor vector ($k \ll N$)
    \item $\lambda_i$: factor loadings for series $i$
    \item $e_{it}$: idiosyncratic component
\end{itemize}
\end{frame}

\section{Estimation}

\begin{frame}{Principal Components}
\textbf{Static approach:} extract factors via PCA on the $T \times N$ data matrix.
\begin{itemize}
    \item \alert{Fast}: eigenvalue decomposition
    \item Consistent as $N, T \to \infty$ (Stock \& Watson, 2002)
    \item Scree plot determines $k$
\end{itemize}

\vspace{0.5em}
\textbf{Dynamic approach:} specify factor dynamics
\[
\mathbf{f}_t = \Phi_1 \mathbf{f}_{t-1} + \cdots + \Phi_s \mathbf{f}_{t-s} + \boldsymbol{\eta}_t
\]
and estimate via \alert{Kalman filter} (state space form).
\end{frame}

\begin{frame}{How Many Factors?}
\begin{itemize}
    \item \textbf{Scree plot}: eigenvalues vs component number --- look for the ``elbow''
    \item \textbf{Bai-Ng criteria}: information criteria adapted for factor models
    \item \textbf{Proportion of variance}: retain components explaining $>$85--90\%
\end{itemize}
\end{frame}

\section{Applications}

\begin{frame}{Factor-Augmented Forecasting}
Use extracted factors as additional predictors:
\[
y_{t+h} = \alpha + \beta_1 y_t + \gamma' \hat{\mathbf{f}}_t + \varepsilon_{t+h}
\]

\textbf{Applications:}
\begin{itemize}
    \item \textbf{Nowcasting}: real-time GDP estimation (mixed frequencies)
    \item \textbf{Monetary policy}: summarising financial conditions
    \item \textbf{FAVAR}: factor-augmented VAR for structural analysis
\end{itemize}

\end{frame}
