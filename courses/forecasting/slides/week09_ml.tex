\section{The Bias-Variance Tradeoff}

\begin{frame}{Why ML for Forecasting?}
\begin{itemize}
    \item Traditional models: low bias, potentially high variance when $p$ is large
    \item \alert{ML insight}: accept some bias to reduce variance $\Rightarrow$ lower total error
    \item Especially useful with \textbf{many predictors} relative to sample size
\end{itemize}

\[
\text{MSE} = \underbrace{\text{Bias}^2}_{\text{model too simple}} + \underbrace{\text{Variance}}_{\text{model too complex}} + \underbrace{\sigma^2_\varepsilon}_{\text{irreducible}}
\]
\end{frame}

\section{Regularised Regression}

\begin{frame}{Ridge and Lasso}
\textbf{Ridge} ($L_2$ penalty):
\[
\hat\beta_{\text{Ridge}} = \arg\min \|y - X\beta\|^2 + \lambda\|\beta\|_2^2
\]
Shrinks all coefficients towards zero.

\vspace{0.5em}
\textbf{Lasso} ($L_1$ penalty):
\[
\hat\beta_{\text{Lasso}} = \arg\min \|y - X\beta\|^2 + \lambda\|\beta\|_1
\]
\alert{Sets some coefficients exactly to zero} --- automatic variable selection!

\vspace{0.5em}
\textbf{Elastic Net}: combines both penalties.
\end{frame}

\begin{frame}{Cross-Validation for $\lambda$}
\begin{enumerate}
    \item Split training data into $K$ folds
    \item For each $\lambda$, compute average out-of-fold MSE
    \item Select $\lambda$ with lowest CV error
\end{enumerate}

\vspace{0.5em}
\alert{Caution for time series:} use \textbf{rolling-window} or \textbf{expanding-window} CV to respect temporal order.
\end{frame}

\section{Tree-Based Methods}

\begin{frame}{Decision Trees and Random Forests}
\textbf{Decision tree:}
\begin{itemize}
    \item Recursively partition the feature space
    \item Predict the mean of each partition
    \item \alert{Nonlinear} and interpretable, but high variance
\end{itemize}

\vspace{0.5em}
\textbf{Random forest:}
\begin{itemize}
    \item Average over $B$ trees, each on a bootstrap sample
    \item Each split considers a random subset of features
    \item \alert{Low variance}, retains nonlinearity
    \item Feature importance: how much does each lag matter?
\end{itemize}
\end{frame}

\begin{frame}{Summary}
\begin{itemize}
    \item \textbf{Bias-variance tradeoff} is the key ML insight
    \item \textbf{Ridge}: shrinks; \textbf{Lasso}: shrinks + selects
    \item \textbf{Random forests}: nonlinear, robust
    \item Use \alert{time-aware cross-validation}
    \item Next week: \alert{neural networks and deep learning}
\end{itemize}
\end{frame}
