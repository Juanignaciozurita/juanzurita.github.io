\section{State Space Representation}

\begin{frame}{The General Model}
\textbf{State equation:}
\[
\boldsymbol{\alpha}_{t+1} = \mathbf{T}\boldsymbol{\alpha}_t + \mathbf{R}\boldsymbol{\eta}_t, \quad \boldsymbol{\eta}_t \sim N(\mathbf{0}, \mathbf{Q})
\]

\textbf{Observation equation:}
\[
\mathbf{y}_t = \mathbf{Z}\boldsymbol{\alpha}_t + \boldsymbol{\varepsilon}_t, \quad \boldsymbol{\varepsilon}_t \sim N(\mathbf{0}, \mathbf{H})
\]

\begin{itemize}
    \item $\boldsymbol{\alpha}_t$: \alert{unobserved} state vector
    \item $\mathbf{y}_t$: observed data
    \item Many time-series models are special cases
\end{itemize}
\end{frame}

\begin{frame}{Why State Space?}
\textbf{Unifying framework:}
\begin{center}
\begin{tabular}{ll}
\toprule
\textbf{Model} & \textbf{State Space Form} \\
\midrule
ARMA & state = lagged values/errors \\
Exponential smoothing & state = level, trend, seasonal \\
Structural time series & state = trend + season + cycle \\
Dynamic factor model & state = latent factors \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\section{The Kalman Filter}

\begin{frame}{Prediction and Update}
\textbf{Prediction step:}
\begin{align}
\hat{\boldsymbol{\alpha}}_{t|t-1} &= \mathbf{T}\hat{\boldsymbol{\alpha}}_{t-1|t-1} \\
\mathbf{P}_{t|t-1} &= \mathbf{T}\mathbf{P}_{t-1|t-1}\mathbf{T}' + \mathbf{R}\mathbf{Q}\mathbf{R}'
\end{align}

\textbf{Update step:}
\begin{align}
\mathbf{K}_t &= \mathbf{P}_{t|t-1}\mathbf{Z}'(\mathbf{Z}\mathbf{P}_{t|t-1}\mathbf{Z}' + \mathbf{H})^{-1} \\
\hat{\boldsymbol{\alpha}}_{t|t} &= \hat{\boldsymbol{\alpha}}_{t|t-1} + \mathbf{K}_t(\mathbf{y}_t - \mathbf{Z}\hat{\boldsymbol{\alpha}}_{t|t-1})
\end{align}

$\mathbf{K}_t$ is the \alert{Kalman gain} --- optimal weighting of new information.
\end{frame}

\section{Structural Time Series}

\begin{frame}{Local Level Model}
\begin{align}
y_t &= \mu_t + \varepsilon_t & \varepsilon_t &\sim N(0, \sigma_\varepsilon^2) \\
\mu_{t+1} &= \mu_t + \eta_t & \eta_t &\sim N(0, \sigma_\eta^2)
\end{align}

\begin{itemize}
    \item Signal-to-noise ratio: $q = \sigma_\eta^2 / \sigma_\varepsilon^2$
    \item $q \to 0$: constant level (all variation is noise)
    \item $q \to \infty$: random walk (all variation is signal)
\end{itemize}
\end{frame}

\begin{frame}{Local Linear Trend}
Adds a stochastic slope:
\begin{align}
\mu_{t+1} &= \mu_t + \nu_t + \eta_t \\
\nu_{t+1} &= \nu_t + \zeta_t
\end{align}

Can add seasonal, cycle, and regression components.

\end{frame}
