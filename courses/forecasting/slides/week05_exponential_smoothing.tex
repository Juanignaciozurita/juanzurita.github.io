\section{Simple Exponential Smoothing}

\begin{frame}{The SES Recursion}
\[
\hat{y}_{t+1|t} = \alpha y_t + (1-\alpha)\hat{y}_{t|t-1}, \quad \alpha \in [0,1]
\]

Equivalently:
\[
\hat{y}_{t+1|t} = \alpha \sum_{j=0}^{\infty} (1-\alpha)^j y_{t-j}
\]

\begin{itemize}
    \item $\alpha$ close to 1: fast adaptation, noisy forecast
    \item $\alpha$ close to 0: slow adaptation, smooth forecast
    \item Optimal for data with \textbf{no trend, no seasonality}
\end{itemize}
\end{frame}

\section{Holt's Linear Trend}

\begin{frame}{Adding a Trend}
Two equations:
\begin{align}
\ell_t &= \alpha y_t + (1-\alpha)(\ell_{t-1} + b_{t-1}) & \text{(level)} \\
b_t &= \beta^*(\ell_t - \ell_{t-1}) + (1-\beta^*)b_{t-1} & \text{(trend)}
\end{align}

\textbf{Forecast:} $\hat{y}_{t+h|t} = \ell_t + h \cdot b_t$

\vspace{0.5em}
\begin{block}{Damped trend variant}
$\hat{y}_{t+h|t} = \ell_t + (\phi + \phi^2 + \cdots + \phi^h) b_t$

Prevents unrealistic long-run extrapolation.
\end{block}
\end{frame}

\section{Holt-Winters}

\begin{frame}{Adding Seasonality}
\textbf{Additive:}
\begin{align}
\ell_t &= \alpha(y_t - s_{t-m}) + (1-\alpha)(\ell_{t-1} + b_{t-1}) \\
b_t &= \beta^*(\ell_t - \ell_{t-1}) + (1-\beta^*)b_{t-1} \\
s_t &= \gamma(y_t - \ell_{t-1} - b_{t-1}) + (1-\gamma)s_{t-m}
\end{align}

Forecast: $\hat{y}_{t+h|t} = \ell_t + hb_t + s_{t+h-m}$

\vspace{0.5em}
\textbf{Multiplicative:} seasonal component multiplies level.
\end{frame}

\section{The ETS Framework}

\begin{frame}{Error, Trend, Seasonal (ETS)}
Systematic taxonomy:
\begin{itemize}
    \item \textbf{Error}: Additive (A) or Multiplicative (M)
    \item \textbf{Trend}: None (N), Additive (A), Damped (Ad)
    \item \textbf{Seasonal}: None (N), Additive (A), Multiplicative (M)
\end{itemize}

\vspace{0.5em}
ETS(A,A,A) = additive Holt-Winters. \\
ETS(M,Ad,M) = damped-trend multiplicative.

\vspace{0.5em}
All can be estimated by \alert{MLE} and compared via \alert{AIC}.
\end{frame}

\begin{frame}{Summary}
\begin{itemize}
    \item \textbf{SES}: no trend, no season --- one parameter $\alpha$
    \item \textbf{Holt}: adds trend --- two parameters $\alpha, \beta^*$
    \item \textbf{Holt-Winters}: adds season --- three parameters $\alpha, \beta^*, \gamma$
    \item \textbf{ETS}: unifying framework; model selection via AIC
    \item Next week: \alert{VAR models} for multivariate forecasting
\end{itemize}
\end{frame}
